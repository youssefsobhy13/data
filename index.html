ما هي SQL؟
lec5 • ص 3-4

تعريف SQL اختصار لـ Structured Query Language — وهي اللغة الأكثر استخداماً للتعامل مع قواعد البيانات العلائقية.

الـ Query (الاستعلام) معناه إنشاء جدول مؤقت يحتوي على نتائج بناءً على شروط معينة من الجداول الأصلية في قاعدة البيانات.

أقسام SQL الثلاثة الرئيسية
DDL — Data Definition Language
إنشاء وتعديل وحذف الجداول والهياكل
CREATE · ALTER · DROP
DML — Data Manipulation Language
إدراج وتعديل وحذف واستعلام البيانات
INSERT · UPDATE · DELETE · SELECT
DCL — Data Control Language
حماية قاعدة البيانات وصلاحيات المستخدمين
GRANT · REVOKE

أنواع البيانات (Data Types)
lec5 • ص 12-13

كل عمود في الجدول لازم يكون له نوع بيانات محدد عشان قاعدة البيانات تعرف تتعامل معاه بشكل صحيح.

النوعالشرحمثال عملي
INTEGER / INTأرقام صحيحة بدون كسورAge INT
SMALLINTأرقام صغيرة (توفير مساحة)code SMALLINT
FLOAT / REALأرقام عشرية تقريبيةprice FLOAT
DECIMAL(i,j)عشري بدقة محددة (i رقم كلي, j كسر)Salary DECIMAL(10,2)
CHAR(n)نص بطول ثابت دائماًGender CHAR(1)
VARCHAR(n)نص بطول متغير (أكثر كفاءة)Name VARCHAR(50)
DATEتاريخ بصيغة YYYY-MM-DDBdate DATE
TIMEوقت بصيغة HH:MM:SSStart_time TIME
TIMESTAMPتاريخ + وقت معاًcreated_at TIMESTAMP
الفرق بين CHAR و VARCHAR: لو اسم طوله 5 حروف وعندك CHAR(15) هياخد 15 مساحة. لو VARCHAR(15) هياخد 5 بس. استخدم VARCHAR دايماً للنصوص.

القيود (Constraints)
lec5 • ص 14-17

القيود هي قوانين إلزامية بنضيفها على الأعمدة لحماية سلامة البيانات ومنع إدخال بيانات خاطئة أو ناقصة.

NOT NULL
يمنع الحقل من أن يكون فارغاً — إجباري الإدخال.
DEFAULT
قيمة افتراضية تُستخدم لو المستخدم ما أدخلش حاجة.
CHECK
شرط يجب تحقيقه — مثل رقم القسم بين 1 و 20.
PRIMARY KEY
المفتاح الأساسي — لا يتكرر ولا يكون NULL، يُعرّف كل سجل بشكل فريد.
UNIQUE
يمنع التكرار لكن يسمح بقيمة NULL واحدة.
FOREIGN KEY
يربط جدول بآخر — يضمن تناسق البيانات بين الجداول.
أمثلة من المحاضرة
Dno INT NOT NULL DEFAULT 1;
Dnumber INT CHECK (Dnumber > 0 AND Dnumber < 21);
PRIMARY KEY (Ssn);
UNIQUE (Dname);
FOREIGN KEY (Super_ssn) REFERENCES EMPLOYEE(Ssn);
FOREIGN KEY (Dno) REFERENCES DEPARTMENT(Dnumber);
لو عندك مشكلة في إنشاء الجداول بسبب FOREIGN KEY، تأكد إنك أنشأت الجدول المُشار إليه الأول.

CREATE TABLE — إنشاء الجداول
lec5 • ص 7-11

أمر CREATE TABLE بنستخدمه لإنشاء جدول جديد. بنحدد اسم كل عمود ونوع بياناته وقيوده.

جدول EMPLOYEE
EMPLOYEE Table — كامل
CREATE TABLE EMPLOYEE (
    Fname        VARCHAR(15) NOT NULL,
    Minit        CHAR,
    Lname        VARCHAR(15) NOT NULL,
    Ssn          CHAR(9)      NOT NULL,
    Bdate        DATE,
    Address      VARCHAR(30),
    Sex          CHAR,
    Salary       DECIMAL(10,2),
    Super_ssn    CHAR(9),
    Dno          INT          NOT NULL,
    PRIMARY KEY (Ssn),
    FOREIGN KEY (Super_ssn) REFERENCES EMPLOYEE(Ssn),
    FOREIGN KEY (Dno)       REFERENCES DEPARTMENT(Dnumber)
);
جدول DEPARTMENT
DEPARTMENT Table
CREATE TABLE DEPARTMENT (
    Dname          VARCHAR(15) NOT NULL,
    Dnumber        INT         NOT NULL,
    Mgr_ssn        CHAR(9)     NOT NULL,
    Mgr_start_date DATE,
    PRIMARY KEY (Dnumber),
    UNIQUE (Dname),
    FOREIGN KEY (Mgr_ssn) REFERENCES EMPLOYEE(Ssn)
);
جدول PROJECT
PROJECT Table
CREATE TABLE PROJECT (
    Pname     VARCHAR(15) NOT NULL,
    Pnumber   INT         NOT NULL,
    Plocation VARCHAR(15),
    Dnum      INT         NOT NULL,
    PRIMARY KEY (Pnumber),
    UNIQUE (Pname),
    FOREIGN KEY (Dnum) REFERENCES DEPARTMENT(Dnumber)
);
جدول WORKS_ON و DEPT_LOCATIONS
WORKS_ON & DEPT_LOCATIONS
CREATE TABLE WORKS_ON (
    Essn   CHAR(9)   NOT NULL,
    Pno    INT        NOT NULL,
    Hours  DECIMAL(3,1) NOT NULL,
    PRIMARY KEY (Essn, Pno),
    FOREIGN KEY (Essn) REFERENCES EMPLOYEE(Ssn),
    FOREIGN KEY (Pno)  REFERENCES PROJECT(Pnumber)
);

CREATE TABLE DEPT_LOCATIONS (
    Dnumber   INT         NOT NULL,
    Dlocation VARCHAR(15) NOT NULL,
    PRIMARY KEY (Dnumber, Dlocation),
    FOREIGN KEY (Dnumber) REFERENCES DEPARTMENT(Dnumber)
);

DML — إدراج وتعديل وحذف
lec5 • ص 18-20
INSERT INTO — إضافة بيانات جديدة
أمثلة INSERT
-- إضافة بأعمدة محددة
INSERT INTO EMPLOYEE (Fname, Lname, Dno, Ssn) 
VALUES ('Richard', 'Marini', 4, '653298653');

-- إضافة كل الأعمدة بالترتيب
INSERT INTO EMPLOYEE
VALUES ('James', 'E', 'Borg', '888665555', '1937-11-10',
        '450 Stone, Houston, TX', 'M', 55000, NULL, 1);
UPDATE — تعديل بيانات موجودة
أمثلة UPDATE
-- تعديل موقع وقسم مشروع معين
UPDATE PROJECT
SET Plocation = 'Bellaire', Dnum = 5
WHERE Pnumber = 10;

-- رفع مرتبات موظفي قسم 5 بنسبة 10%
UPDATE EMPLOYEE
SET Salary = Salary * 1.1
WHERE Dno = 5;
DELETE — حذف البيانات
أمثلة DELETE
-- حذف موظف معين
DELETE FROM EMPLOYEE WHERE Lname = 'Brown';

-- حذف كل الموظفين في قسم 5
DELETE FROM EMPLOYEE WHERE Dno = 5;

-- حذف كل الصفوف (خطر!)
DELETE FROM EMPLOYEE;
DELETE بدون WHERE هيحذف كل البيانات من الجدول! دايماً تأكد من وجود WHERE.

SELECT — استعلام البيانات
lec5 • ص 21-31

الأهم SELECT هو الأمر الأكثر استخداماً في SQL — بيجيب البيانات من جدول أو أكثر.

SELECT الأساسي
أمثلة أساسية
-- جيب اسم الموظف وعنوانه لشخص اسمه John
SELECT Fname, Address
FROM EMPLOYEE
WHERE Fname = 'John';

-- جيب كل حاجة من الجدول
SELECT * FROM EMPLOYEE;

-- إزالة التكرار
SELECT DISTINCT Salary FROM EMPLOYEE;
LIKE — البحث بنمط
LIKE مع % و _
-- % = أي عدد من الحروف
SELECT Fname, Lname
FROM EMPLOYEE
WHERE Address LIKE '%Houston,TX%';

-- _ = حرف واحد بالضبط
SELECT Fname FROM EMPLOYEE WHERE Fname LIKE '__n%';
-- يجيب الأسماء اللي حرفها الثالث n
BETWEEN و IN
BETWEEN & IN
-- المرتبات بين 30000 و 40000 في قسم 5
SELECT * FROM EMPLOYEE
WHERE (Salary BETWEEN 30000 AND 40000) AND Dno = 5;

-- موظفو أقسام 1 و 5
SELECT * FROM EMPLOYEE
WHERE Dno IN (1, 5);
ORDER BY و الدوال التجميعية
ORDER BY & Aggregate Functions
-- ترتيب تنازلي بالمرتب
SELECT * FROM EMPLOYEE ORDER BY Salary DESC;

-- دوال تجميعية
SELECT COUNT(*) FROM EMPLOYEE;             -- عدد الصفوف
SELECT AVG(Salary) FROM EMPLOYEE;          -- متوسط المرتب
SELECT MAX(Salary) FROM EMPLOYEE;          -- أعلى مرتب
SELECT MIN(Salary) FROM EMPLOYEE;          -- أقل مرتب
SELECT SUM(Salary) FROM EMPLOYEE;          -- مجموع المرتبات

-- GROUP BY — عدد الموظفين لكل قسم
SELECT Dno, COUNT(*) AS NumEmployees
FROM EMPLOYEE
GROUP BY Dno;

-- HAVING — تصفية بعد GROUP BY
SELECT Dno, AVG(Salary)
FROM EMPLOYEE
GROUP BY Dno
HAVING AVG(Salary) > 40000;
JOIN — ربط الجداول
JOIN Examples
-- جيب اسم الموظف واسم قسمه
SELECT E.Fname, E.Lname, D.Dname
FROM EMPLOYEE E, DEPARTMENT D
WHERE E.Dno = D.Dnumber;

-- بنفس النتيجة باستخدام JOIN
SELECT E.Fname, D.Dname
FROM EMPLOYEE E
JOIN DEPARTMENT D ON E.Dno = D.Dnumber;

CompanyDB — Full Script
DROP DATABASE IF EXISTS CompanyDB;
CREATE DATABASE CompanyDB;
USE CompanyDB;

CREATE TABLE Department (
    Dnumber INT PRIMARY KEY,
    Dname   VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE Employee (
    SSN    INT PRIMARY KEY,

What is SQL?
lec5 • page 3-4

Definition SQL stands for Structured Query Language — the most widely used language for interacting with relational databases.

A Query is creating a temporary derived table containing specific results based on conditions we specify from the original database tables.

SQL Three Main Divisions
DDL — Data Definition Language
Create, alter, and drop tables and structures
CREATE · ALTER · DROP
DML — Data Manipulation Language
Insert, update, delete, and query data
INSERT · UPDATE · DELETE · SELECT
DCL — Data Control Language
Protect the database and manage user permissions
GRANT · REVOKE
CREATE DATABASE
CREATE DATABASE COMPANY;
This command creates a new database called COMPANY. After creation, use USE COMPANY; to start working with it.

Data Types
lec5 • page 12-13

Every column in a table must have a specified data type so the database knows how to handle it.

TypeDescriptionExample
INTEGER / INTWhole numbers without decimalsAge INT
FLOAT / REALApproximate decimal numbersprice FLOAT
DECIMAL(i,j)Exact decimal (i total digits, j after point)Salary DECIMAL(10,2)
CHAR(n)Fixed-length string — always uses n charsGender CHAR(1)
VARCHAR(n)Variable-length string — more efficientName VARCHAR(50)
DATEDate in YYYY-MM-DD formatBdate DATE
TIMETime in HH:MM:SS formatStart_time TIME
TIMESTAMPDate + Time combinedcreated_at TIMESTAMP
CHAR vs VARCHAR: CHAR(15) always stores 15 characters even if name is only 5. VARCHAR(15) stores only what's needed. Prefer VARCHAR for text.

Constraints
lec5 • page 14-17

Constraints are mandatory rules added to columns to ensure data integrity and prevent invalid or incomplete data entry.

NOT NULL
Column cannot be empty — input is required.
DEFAULT
A default value is used if the user provides none.
CHECK
A condition data must satisfy — e.g. Dnumber between 1 and 20.
PRIMARY KEY
Uniquely identifies each record — cannot be NULL or duplicate.
UNIQUE
Prevents duplicate values but allows one NULL.
FOREIGN KEY
Links one table to another — ensures referential integrity.
Constraint Examples
Dno INT NOT NULL DEFAULT 1;
Dnumber INT CHECK (Dnumber > 0 AND Dnumber < 21);
PRIMARY KEY (Ssn);
UNIQUE (Dname);
FOREIGN KEY (Dno) REFERENCES DEPARTMENT(Dnumber);

LIKE, BETWEEN, ORDER BY, Aggregates

SELECT * FROM Employee WHERE Salary BETWEEN 3000 AND 5000;

SELECT * FROM Employee ORDER BY Salary DESC;
