🏢 CompanyDB قاعدة بيانات المثال

إنشاء الجداول (DDL)
CREATE TABLE Department (
    Dnumber INT PRIMARY KEY,
    Dname VARCHAR(50) NOT NULL
);

CREATE TABLE Employee (
    SSN INT PRIMARY KEY,
    Fname VARCHAR(50),
    Salary FLOAT,
    Dno INT,
    FOREIGN KEY (Dno) REFERENCES Department(Dnumber)
);

CREATE TABLE Works_On (
    Essn INT,
    Pno INT,
    Hours FLOAT,
    PRIMARY KEY (Essn, Pno)
);

✅ حلول الأسئلة Exam Questions

Q1: أسماء تبدأ بـ A
SELECT * FROM Employee WHERE Fname LIKE 'A%';
Q2: مرتب بين 3500 و 5500
SELECT * FROM Employee WHERE Salary BETWEEN 3500 AND 5500;
Q3: أسماء موظفين وأقسام
SELECT Employee.Fname, Department.Dname 
FROM Employee, Department 
WHERE Employee.Dno = Department.Dnumber;
Q4: إدراج قسم جديد
INSERT INTO Department VALUES (4, 'Finance');
Q6: متوسط المرتبات
SELECT AVG(Salary) FROM Employee;
Q17: ترتيب تنازلي حسب المرتب
SELECT * FROM Employee ORDER BY Salary DESC;
متوسط المرتب لكل قسم
SELECT Dno, AVG(Salary) FROM Employee GROUP BY Dno;
